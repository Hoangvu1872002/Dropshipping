  <script class="rr-ba-script">
    (function () {
      window._rrBaLoaded = true;
      function getLimits(canvas, root) {
        var beforeBadge = root.querySelector('.rr-ba-badge-before');
        var afterBadge = root.querySelector('.rr-ba-badge-after');
        var rect = canvas.getBoundingClientRect();
        var minP = 14;
        var maxP = 86;

        if (rect && rect.width > 0) {
          if (beforeBadge) {
            var bRect = beforeBadge.getBoundingClientRect();
            var minPx = (bRect.right - rect.left) + 28;
            minP = Math.max((minPx / rect.width) * 100, 10);
          }
          if (afterBadge) {
            var aRect = afterBadge.getBoundingClientRect();
            var maxPx = (aRect.left - rect.left) - 28;
            maxP = Math.min((maxPx / rect.width) * 100, 90);
          }
        }
        return { min: minP, max: maxP };
      }

      var rafId = null;
      function setSliderPos(canvas, root, clientX) {
        if (!canvas || !root) return;
        if (rafId) cancelAnimationFrame(rafId);
        rafId = requestAnimationFrame(function () {
          var rect = canvas.getBoundingClientRect();
          if (!rect || rect.width <= 0) return;
          var offset = clientX - rect.left;
          var percent = (offset / rect.width) * 100;
          var limits = getLimits(canvas, root);
          var clamped = Math.min(Math.max(percent, limits.min), limits.max);
          root.style.setProperty('--rr-ba-pos', clamped + '%');

          var range = root.querySelector('.rr-ba-range');
          if (range) range.value = clamped;

          var hint = root.querySelector('.rr-ba-hint');
          if (hint) hint.classList.add('is-hidden');
        });
      }

      function setActiveState(canvas, root, on) {
        if (!canvas) return;
        var handle = root ? root.querySelector('.rr-ba-handle') : null;
        if (on) {
          canvas.classList.add('is-active', 'is-dragging', 'is-focused');
          if (handle) handle.classList.add('is-active');
        } else {
          canvas.classList.remove('is-active', 'is-dragging', 'is-focused');
          if (handle) handle.classList.remove('is-active');
        }
      }

      var activeCanvas = null;
      var activeRoot = null;
      var isDragging = false;
      var currentPointerId = null;

      function resetDrag() {
        if (isDragging && activeCanvas && currentPointerId !== null) {
          try {
            if (activeCanvas.releasePointerCapture) {
              activeCanvas.releasePointerCapture(currentPointerId);
            }
          } catch (e) { }
        }
        if (activeCanvas && activeRoot) {
          setActiveState(activeCanvas, activeRoot, false);
          var range = activeRoot.querySelector('.rr-ba-range');
          if (range) {
            try { range.blur(); } catch (e) { }
          }
        }
        isDragging = false;
        currentPointerId = null;
        activeCanvas = null;
        activeRoot = null;
      }

      if (!window._rrBaGlobalDelegationAttached) {
        window._rrBaGlobalDelegationAttached = true;

        document.addEventListener('pointerdown', function (e) {
          var canvas = e.target.closest('.rr-ba-canvas');
          if (!canvas) return;
          var root = canvas.closest('.rr-ba-wrapper') || canvas.parentElement;
          if (!root) return;

          isDragging = true;
          activeCanvas = canvas;
          activeRoot = root;
          currentPointerId = e.pointerId;

          setActiveState(canvas, root, true);

          try {
            if (typeof e.pointerId === 'number' && canvas.setPointerCapture) {
              canvas.setPointerCapture(e.pointerId);
            }
          } catch (err) { }

          if (typeof e.clientX === 'number') {
            setSliderPos(canvas, root, e.clientX);
          }
        }, { passive: true });

        document.addEventListener('pointermove', function (e) {
          if (!isDragging || !activeCanvas || !activeRoot) return;
          if (typeof e.clientX === 'number') {
            setSliderPos(activeCanvas, activeRoot, e.clientX);
          }
        }, { passive: true });

        document.addEventListener('pointerup', resetDrag, { passive: true });
        document.addEventListener('pointercancel', resetDrag, { passive: true });
        document.addEventListener('lostpointercapture', resetDrag);
        window.addEventListener('pointerup', resetDrag, { passive: true });
        window.addEventListener('mouseup', resetDrag, { passive: true });
        window.addEventListener('touchend', resetDrag, { passive: true });

        window.addEventListener('blur', resetDrag);
        document.addEventListener('visibilitychange', function () {
          if (document.hidden) resetDrag();
        });
        window.addEventListener('pagehide', resetDrag);
        window.addEventListener('pageshow', resetDrag);

        document.addEventListener('input', function (e) {
          var range = e.target.closest('.rr-ba-range');
          if (!range) return;
          var root = range.closest('.rr-ba-wrapper');
          var canvas = root ? root.querySelector('.rr-ba-canvas') : null;
          if (!root || !canvas) return;

          var val = parseFloat(range.value);
          var limits = getLimits(canvas, root);
          var clamped = Math.min(Math.max(val, limits.min), limits.max);
          root.style.setProperty('--rr-ba-pos', clamped + '%');
          range.value = clamped;

          var hint = root.querySelector('.rr-ba-hint');
          if (hint) hint.classList.add('is-hidden');
        });

        document.addEventListener('focusin', function (e) {
          var range = e.target.closest('.rr-ba-range');
          if (!range) return;
          var root = range.closest('.rr-ba-wrapper');
          var canvas = root ? root.querySelector('.rr-ba-canvas') : null;
          if (canvas && root) setActiveState(canvas, root, true);
        });

        document.addEventListener('focusout', function (e) {
          var range = e.target.closest('.rr-ba-range');
          if (!range) return;
          var root = range.closest('.rr-ba-wrapper');
          var canvas = root ? root.querySelector('.rr-ba-canvas') : null;
          if (canvas && root) setActiveState(canvas, root, false);
        });
      }

      function initAllSliders() {
        resetDrag();
        var allWrappers = document.querySelectorAll('.rr-ba-wrapper');
        allWrappers.forEach(function (root) {
          var pos = root.style.getPropertyValue('--rr-ba-pos') || '50%';
          var range = root.querySelector('.rr-ba-range');
          if (range && (!range.value || range.value === '50')) {
            range.value = parseFloat(pos) || 50;
          }
        });
      }

      window._rrBaInit = initAllSliders;
      initAllSliders();

      window.addEventListener('pageshow', initAllSliders);
      window.addEventListener('popstate', initAllSliders);
      document.addEventListener('DOMContentLoaded', initAllSliders);
      document.addEventListener('turbo:load', initAllSliders);
      document.addEventListener('turbolinks:load', initAllSliders);
      document.addEventListener('shopify:section:load', initAllSliders);
      document.addEventListener('pagefly:section:load', initAllSliders);
    })();
  </script>
